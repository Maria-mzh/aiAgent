### 2.3 数据迁移方案

#### 新增表迁移
```sql
-- 创建新表
CREATE TABLE `new_table` (...);

-- 初始化数据（如需要）
INSERT INTO `new_table` SELECT ... FROM `old_table`;
```

#### 数据变更迁移
```sql
-- 备份原表
CREATE TABLE `table_name_backup_20240101` LIKE `table_name`;
INSERT INTO `table_name_backup_20240101` SELECT * FROM `table_name`;

-- 添加新字段
ALTER TABLE `table_name` ADD COLUMN `new_field` varchar(100) COMMENT '新字段';

-- 数据迁移
UPDATE `table_name` SET `new_field` = 'default_value' WHERE `new_field` IS NULL;
```

#### 回滚方案
```sql
-- 回滚表结构变更
ALTER TABLE `table_name` DROP COLUMN `new_field`;

-- 恢复数据
TRUNCATE TABLE `table_name`;
INSERT INTO `table_name` SELECT * FROM `table_name_backup_20240101`;
```
